SELECT
  f.festival_year       AS Έτος,
  COALESCE(t.purchase_method, 'Σύνολο_Εσόδων') AS Τρόπος_Πληρωμής,
  SUM(t.ticket_cost)    AS Σύνολο_Εσόδων
FROM Ticket    AS t
JOIN Event     AS e  ON t.event_id_FK    = e.event_id
JOIN Festival  AS f  ON e.festival_id_FK = f.festival_id
GROUP BY
  f.festival_year,
  t.purchase_method
WITH ROLLUP
HAVING
  (festival_year IS NOT NULL AND purchase_method IS NOT NULL)
  OR
  (festival_year IS NOT NULL AND purchase_method IS NULL)
ORDER BY
  f.festival_year,
  purchase_method IS NULL,
  purchase_method;

/* Δώστε τιμές στις δύο μεταβλητές πριν τρέξετε το SELECT */
SET @p_genre = 'rock';   
SET @p_year  = 2024;     

SELECT  a.artist_id,
        COALESCE(a.pseudonym, a.real_name) AS artist_name,
        a.music_genre,
        /* ‘Y’ αν ο καλλιτέχνης έπαιξε σε εκδήλωση τη συγκεκριμένη χρονιά */
        CASE WHEN x.artist_id_FK IS NOT NULL THEN 'Y' ELSE 'N' END AS participated
FROM    Artist AS a
LEFT JOIN (   /* σύνολο καλλιτεχνών που εμφανίστηκαν το @p_year */
        SELECT DISTINCT pa.artist_id_FK
        FROM   Performance_Artist pa
        JOIN   Performance       p ON p.performance_id = pa.performance_id_FK
        JOIN   Event             e ON e.event_id        = p.event_id_FK
        JOIN   Festival          f ON f.festival_id     = e.festival_id_FK
        WHERE  f.festival_year = @p_year
) AS x  ON x.artist_id_FK = a.artist_id
WHERE   a.music_genre = @p_genre           
ORDER BY artist_name;

SELECT  a.artist_id,
        COALESCE(a.pseudonym, a.real_name) AS artist_name,
        f.festival_id,
        f.festival_name,
        f.festival_year,
        COUNT(DISTINCT p.performance_id) AS warmup_cnt              
FROM        Artist              AS a
JOIN        Performance_Artist  AS pa ON pa.artist_id_FK   = a.artist_id
JOIN        Performance         AS p  ON p.performance_id  = pa.performance_id_FK
JOIN        Event               AS e  ON e.event_id        = p.event_id_FK
JOIN        Festival            AS f  ON f.festival_id     = e.festival_id_FK
WHERE       p.performance_type = 'warm up'
GROUP BY    a.artist_id, f.festival_id                           
HAVING      warmup_cnt >= 3                                      
ORDER BY    warmup_cnt DESC, artist_name;

    SUM(
        r.rating_interpretation
      + r.rating_sound_light
      + r.rating_stage_presence
      + r.rating_organization
      + r.rating_overall
    )                                         AS total_score
FROM   Rating              AS r
JOIN   Visitor             AS v  ON v.visitor_id           = r.visitor_id_FK
JOIN   Performance         AS p  ON p.performance_id       = r.performance_id_FK
JOIN   Performance_Artist  AS pa ON pa.performance_id_FK   = p.performance_id
JOIN   Artist              AS a  ON a.artist_id            = pa.artist_id_FK
GROUP  BY v.visitor_id, a.artist_id, visitor_name, artist_name
ORDER  BY total_score DESC
LIMIT  5;
