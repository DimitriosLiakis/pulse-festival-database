\section*{2.1 Απαραίτητοι Περιορισμοί για την Ορθότητα της ΒΔ}

\subsection*{1. Ακεραιότητα Οντοτήτων (Entity Integrity)}
\begin{itemize}
  \item \textbf{PRIMARY KEY:}  
  \begin{itemize}
    \item Location(location\_id)
    \item Festival(festival\_id)
    \item Stage(stage\_id)
    \item Event(event\_id)
    \item Artist(artist\_id)
    \item Band(band\_id)
    \item Artist\_Band(artist\_id, band\_id)
    \item Performance(performance\_id)
    \item Performance\_Artist(performance\_id, artist\_id)
    \item Performance\_Band(performance\_id, band\_id)
    \item Visitor(visitor\_id)
    \item Ticket(ticket\_id)
    \item Staff(staff\_id)
    \item Stage\_Staff(stage\_id, staff\_id, start\_date)
    \item Rating(rating\_id)
    \item Resale\_queue(resale\_id)
  \end{itemize}
\end{itemize}

    \item Performance\_Band.performance\_id $\to$ Performance.performance\_id (ON DELETE CASCADE)
    \item Performance\_Band.band\_id $\to$ Band.band\_id (ON DELETE CASCADE)

\subsection*{3. Ακεραιότητα Πεδίου Τιμών (Domain Integrity)}
\subsubsection*{3.1 Check Constraints}
\begin{itemize}
  \item \textbf{Festival:} \texttt{CHECK (end\_date >= start\_date)}
  \item \textbf{Stage:} \texttt{CHECK (capacity > 0)}
  \item \textbf{Event:} \texttt{CHECK (event\_end > event\_start)}
  \item \textbf{Performance:}
    \begin{itemize}
      \item \texttt{CHECK (performance\_type IN ('warm up','headline','special guest','main act'))}
      \item \texttt{CHECK (duration\_minutes BETWEEN 1 AND 180)}
    \end{itemize}
  \item \textbf{Visitor:} \texttt{CHECK (age BETWEEN 0 AND 120)}
  \item \textbf{Staff:}
    \begin{itemize}
      \item \texttt{CHECK (age >= 18)}
      \item \texttt{CHECK (role IN ('tech','security','support'))}
      \item \texttt{CHECK (experience IN ('intern','beginner','average','experienced','expert'))}
    \end{itemize}
  \item \textbf{Ticket:}
    \begin{itemize}
      \item \texttt{CHECK (ticket\_cost >= 0)}
      \item \texttt{CHECK (ticket\_category IN ('general','VIP','backstage'))}
      \item \texttt{CHECK (purchase\_method IN ('credit\_card','debit\_card','bank\_transfer','other'))}
    \end{itemize}
  \item \textbf{Rating:}
    \begin{itemize}
      \item \texttt{CHECK (rating\_interpretation BETWEEN 1 AND 5)}
      \item \texttt{CHECK (rating\_sound\_light BETWEEN 1 AND 5)}
      \item \texttt{CHECK (rating\_stage\_presence BETWEEN 1 AND 5)}
      \item \texttt{CHECK (rating\_organization BETWEEN 1 AND 5)}
      \item \texttt{CHECK (rating\_overall BETWEEN 1 AND 5)}
    \end{itemize}
  \item \textbf{Stage\_Staff:} \texttt{CHECK (end\_date >= start\_date)}
  \item \textbf{Resale\_queue:} \texttt{CHECK (interest\_type IN ('seller','buyer'))}
\end{itemize}

\subsubsection*{3.2 Μήκη και Τύποι Πεδίων}
\begin{itemize}
  \item \texttt{VARCHAR} και \texttt{CHAR(13)} με περιορισμένο μήκος.
  \item Τύποι \texttt{DATE}, \texttt{DATETIME}, \texttt{YEAR} για σωστή αποθήκευση χρόνου.
\end{itemize}

\subsection*{4. Μοναδικότητα (Uniqueness)}
\begin{itemize}
  \item \textbf{Ticket:}
    \begin{itemize}
      \item \texttt{UNIQUE (ean\_code)}
      \item \texttt{UNIQUE (visitor\_id\_FK, event\_id\_FK)}
    \end{itemize}
  \item \textbf{Artist\_Band}, \textbf{Performance\_Artist}, \textbf{Performance\_Band}, \textbf{Stage\_Staff}, \textbf{Rating}: σύνθετα PK.
\end{itemize}

\subsection*{5. Περιορισμοί Οριζόμενοι από τον Χρήστη (Triggers)}
\begin{enumerate}
  \item \textbf{trg\_staff\_quota} (BEFORE INSERT ON Stage\_Staff):
    Ελέγχει security ≥ 5\% και support ≥ 2\% της χωρητικότητας, στρογγυλοποίηση προς τα πάνω.
  \item \textbf{trg\_ticket\_capacity\_vip} (BEFORE INSERT ON Ticket):
    Ελέγχοι: συνολική χωρητικότητα και VIP ≤ 10\% capacity.
  \item \textbf{trg\_performance\_break} (BEFORE INSERT/UPDATE ON Performance):
    Εξασφάλιση διαλείμματος 5–30 λεπτών μεταξύ διαδοχικών εμφανίσεων.
  \item \textbf{trg\_artist\_overlap} (BEFORE INSERT ON Performance\_Artist/Performance\_Band):
    Αποφυγή χρονικής επικάλυψης εμφανίσεων.
  \item \textbf{trg\_consecutive\_years} (BEFORE INSERT ON Performance\_Artist/Performance\_Band):
    Περιορισμός σε ≤ 3 συνεχόμενα έτη συμμετοχής.
\end{enumerate}

\subsection{Ευρετήρια}

Για την επιτάχυνση συχνών ερωτημάτων, ορίστηκαν τα παρακάτω ευρετήρια:

\begin{itemize}
  \item \textbf{FK indexes}  
    \begin{lstlisting}[language=SQL]
CREATE INDEX idx_event_festival   ON Event(festival_id_FK);
CREATE INDEX idx_event_stage      ON Event(stage_id_FK);
CREATE INDEX idx_perf_event       ON Performance(event_id_FK);
CREATE INDEX idx_pa_artist        ON Performance_Artist(artist_id);
CREATE INDEX idx_perfband_band    ON Performance_Band(band_id);
CREATE INDEX idx_ticket_event     ON Ticket(event_id_FK);
CREATE INDEX idx_ticket_visitor   ON Ticket(visitor_id_FK);
CREATE INDEX idx_rating_perf      ON Rating(performance_id_FK);
CREATE INDEX idx_rating_visitor   ON Rating(visitor_id_FK);
CREATE INDEX idx_stage_staff_st   ON Stage_Staff(stage_id);
CREATE INDEX idx_stage_staff_staff ON Stage_Staff(staff_id);
    \end{lstlisting}
    \emph{Χρήση:} Επιταχύνουν JOINs σε συσχετισμούς Performance→Event, Event→Festival, Rating→Performance, κ.ά.
  \item \textbf{Χρονικά ευρετήρια}  
    \begin{lstlisting}[language=SQL]
CREATE INDEX idx_event_period     ON Event(event_start, event_end);
CREATE INDEX idx_performance_time ON Performance(start_time);
    \end{lstlisting}
    \emph{Χρήση:} Εύρεση γεγονότων/εμφανίσεων σε χρονικά διαστήματα (π.χ. Q1, Q3, Q6).
  \item \textbf{Composite ευρετήριο για μοναδικότητα εισιτηρίων}  
    Το UNIQUE KEY \texttt{uk\_ticket\_per\_day (visitor\_id\_FK, event\_id\_FK)} στον πίνακα \texttt{Ticket} εξασφαλίζει ότι κάθε επισκέπτης έχει το πολύ ένα εισιτήριο ανά παράσταση.
\end{itemize}

DROP DATABASE IF EXISTS pko;

  CHARACTER SET utf8mb4

USE pko;

CREATE TABLE Location (
  location_id INT PRIMARY KEY,
  address VARCHAR(255) NOT NULL,
  city VARCHAR(100) NOT NULL,

  continent VARCHAR(50) NOT NULL,

  location_image VARCHAR(255)

CREATE TABLE Festival (
  festival_id INT PRIMARY KEY,
  festival_name VARCHAR(100) NOT NULL,

  location_id_FK INT NOT NULL,
  festival_image VARCHAR(255),
  FOREIGN KEY (location_id_FK)
    REFERENCES Location(location_id)
      ON UPDATE CASCADE
      ON DELETE RESTRICT,

CREATE TABLE Stage (
  stage_id INT PRIMARY KEY,
  stage_name VARCHAR(100) NOT NULL,
  stage_description TEXT,
  capacity INT NOT NULL CHECK (capacity > 0),

  stage_image VARCHAR(255)

CREATE TABLE Event (
  event_id INT PRIMARY KEY,
  event_name VARCHAR(100) NOT NULL,
  festival_id_FK INT NOT NULL,
  stage_id_FK INT NOT NULL,
  event_start DATETIME NOT NULL,
  event_end DATETIME NOT NULL,
  event_image VARCHAR(255),
  FOREIGN KEY (festival_id_FK)

      ON DELETE CASCADE,
  FOREIGN KEY (stage_id_FK)

CREATE TABLE Artist (

  FOREIGN KEY (stage_id)

      ON DELETE CASCADE,
  FOREIGN KEY (staff_id)

CREATE TABLE Rating (
  rating_id INT PRIMARY KEY,

  rating_sound_light    TINYINT NOT NULL

  rating_stage_presence TINYINT NOT NULL

    CHECK (rating_overall BETWEEN 1 AND 5),

  FOREIGN KEY (performance_id_FK)

CREATE TABLE Resale_queue (
  resale_id INT PRIMARY KEY,

  interest_date DATETIME NOT NULL,
  FOREIGN KEY (ticket_id_FK)
    REFERENCES Ticket(ticket_id)
      ON UPDATE CASCADE
      ON DELETE CASCADE,
  FOREIGN KEY (visitor_id_FK)

CREATE INDEX idx_event_period

DROP TRIGGER IF EXISTS trg_staff_quota;
DELIMITER $$
CREATE TRIGGER trg_staff_quota
BEFORE INSERT ON Stage_Staff
FOR EACH ROW
BEGIN
  DECLARE cap INT;
  DECLARE role_txt VARCHAR(30);
  DECLARE cnt INT;
  IF IFNULL(@bulk_load,0)=0 THEN
    SELECT capacity INTO cap
      FROM Stage
     WHERE stage_id = NEW.stage_id;

    SELECT role INTO role_txt
      FROM Staff
     WHERE staff_id = NEW.staff_id;

    SELECT COUNT(*) INTO cnt
      FROM Stage_Staff ss
      JOIN Staff s ON s.staff_id = ss.staff_id
     WHERE ss.stage_id = NEW.stage_id
       AND ss.start_date = NEW.start_date
       AND s.role = role_txt;

    IF role_txt = 'security'
       AND cnt + 1 < CEIL(cap * 0.05) THEN
      SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Security staff below 5%';
    ELSEIF role_txt = 'support'
       AND cnt + 1 < CEIL(cap * 0.02) THEN
      SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Support staff below 2%';
    END IF;
  END IF;
END$$

\subsection{Ερώτημα 4: Μέσος Όρος Αξιολογήσεων Καλλιτέχνη}

Για τον καλλιτέχνη με \verb|artist_id = 5| εκτελέστηκαν τα ακόλουθα ερωτήματα και αναλύθηκαν τρεις στρατηγικές join:

\begin{enumerate}
  \item \textbf{Βασικό Query}  
  \begin{verbatim}
SELECT
  a.artist_id,
  a.name                     AS artist_name,
  AVG(r.rating_interpretation) AS avg_interpretation,
  AVG(r.rating_overall)        AS avg_overall
FROM Rating             AS r
JOIN Performance_Artist AS pa
  ON r.performance_id_FK = pa.performance_id
JOIN Artist             AS a
  ON pa.artist_id       = a.artist_id
WHERE a.artist_id = 5
GROUP BY a.artist_id;
  \end{verbatim}
  \textit{Αποτελέσματα:}
  \begin{tabular}{l l r r}
    \textbf{artist\_id} & \textbf{artist\_name} & \textbf{avg\_interpretation} & \textbf{avg\_overall} \\
    \hline
    5 & William Herrera & 3.8182 & 3.2727
  \end{tabular}

  \smallskip
  \textit{EXPLAIN FORMAT=JSON (default plan): cost = 4.05}
  \begin{verbatim}
"nested_loop": [
  { "table_name": "a", "access_type": "const", "key": "PRIMARY", … },
  { "table_name": "pa","access_type": "ref",   "key": "artist_id", … },
  { "table_name": "r", "access_type": "ref",   "key": "performance_id_FK", … }
]
  \end{verbatim}

  \smallskip
  \textit{Optimizer Trace (key steps):}
  \begin{verbatim}
"strategy": "ref_or_null" on r (2500 rows),
then "unique_subquery" on pa and a (1 row each).
  \end{verbatim}

  \item \textbf{Forced Nested Loop Join (4.3)}  
  \begin{verbatim}
/*+ LOOP_JOIN(r pa) */
SELECT … 
FROM Rating AS r
  FORCE INDEX (performance_id_FK)
JOIN Performance_Artist AS pa
  FORCE INDEX (PRIMARY)
  /*+ LOOP_JOIN(r pa) */
  ON r.performance_id_FK = pa.performance_id
JOIN Artist AS a
  FORCE INDEX (PRIMARY)
  ON pa.artist_id = a.artist_id
WHERE a.artist_id = 5
GROUP BY a.artist_id;
  \end{verbatim}
  \textit{EXPLAIN FORMAT=JSON: cost = 27.22}
  \begin{verbatim}
"nested_loop": [
  { "table_name": "a", "access_type": "const", "key": "PRIMARY", … },
  { "table_name": "pa","access_type": "index", "key": "PRIMARY", … },
  { "table_name": "r", "access_type": "ref",   "key": "performance_id_FK", … }
]
  \end{verbatim}

  \item \textbf{Forced Hash Join (4.4)}  
  \begin{verbatim}

/*+ HASH_JOIN(r pa) */
SELECT …
FROM Rating AS r
  FORCE INDEX (performance_id_FK)
JOIN Performance_Artist AS pa
  FORCE INDEX (PRIMARY)
  /*+ HASH_JOIN(r pa) */
  ON r.performance_id_FK = pa.performance_id
JOIN Artist AS a
  FORCE INDEX (PRIMARY)
  ON pa.artist_id = a.artist_id
WHERE a.artist_id = 5
GROUP BY a.artist_id;
  \end{verbatim}
  \textit{EXPLAIN FORMAT=JSON: cost ≃ 27.22}
  \begin{verbatim}

  { "strategy": "build_hash_table", "table": "pa", "rows": 68 },
  { "strategy": "probe_hash",       "table": "r",  "rows": 4  }
]
  \end{verbatim}

  \item \textbf{Forced Index Only}  
  \begin{verbatim}

FROM Rating AS r
  FORCE INDEX (performance_id_FK)
JOIN Performance_Artist AS pa
  FORCE INDEX (PRIMARY)
  ON …
JOIN Artist AS a
  FORCE INDEX (PRIMARY)
  ON …
WHERE a.artist_id = 5

  \end{verbatim}
  \textit{EXPLAIN FORMAT=JSON: key for r = performance\_id\_FK, for pa/a = PRIMARY}

\end{enumerate}

\paragraph{Συμπεράσματα}
\begin{itemize}
  \item \textbf{Default Nested Loop} είναι το πιο γρήγορο (cost=4.05) καθώς το \texttt{Performance\_Artist} είναι μικρό.  
  \item \textbf{Forced NL} αυξάνει το cost (27.22) λόγω αναγκαστικών index‐scans στο \texttt{pa}.  
  \item \textbf{Hash Join} δεν ωφελεί σημαντικά με το τρέχον μέγεθος (παρόμοιο cost).  
  \item \textbf{Forced Index Only} επιβεβαιώνει χρήση των επιλεγμένων indexes χωρίς αλλαγή join method.  
\end{itemize}

\subsection{Ερώτημα 6: Παραστάσεις Επισκέπτη και Μέσος Όρος Βαθμολογίας}

Για τον επισκέπτη με \verb|visitor_id = 24| εκτελέστηκαν τα ακόλουθα ερωτήματα και αναλύθηκαν τρεις στρατηγικές join καθώς και forced index:

\begin{enumerate}
  \item \textbf{Βασικό Query}  
  \begin{verbatim}
SELECT
  p.performance_id,
  e.event_name,
  p.performance_type,
  AVG(
    (r.rating_interpretation
   + r.rating_sound_light
   + r.rating_stage_presence
   + r.rating_organization
   + r.rating_overall)
    / 5.0
  ) AS avg_visitor_rating
FROM Rating      AS r
JOIN Performance AS p
  ON r.performance_id_FK = p.performance_id
JOIN Event       AS e
  ON p.event_id_FK      = e.event_id
WHERE r.visitor_id_FK = 24
GROUP BY
  p.performance_id,
  e.event_name,
  p.performance_type
ORDER BY
  e.event_name,
  p.start_time;
  \end{verbatim}
  \textit{Αποτελέσματα:}
  \begin{tabular}{r l l}
    \textbf{performance\_id} & \textbf{event\_name} & \textbf{avg\_visitor\_rating} \\
    \hline
    42 & Event\_23 & 3.6000 \\
    19 & Event\_36 & 3.2000 \\
    27 & Event\_49 & 3.0000 \\
    23 & Event\_9  & 3.8000
  \end{tabular}

  \smallskip
  \textit{EXPLAIN FORMAT=JSON (default plan): cost = 3.95}
  \begin{verbatim}
"nested_loop": [
  { "table_name":"r","access_type":"ref","key":"uk_rating_once",… },
  { "table_name":"p","access_type":"eq_ref","key":"PRIMARY",… },
  { "table_name":"e","access_type":"eq_ref","key":"PRIMARY",… }
]
  \end{verbatim}

  \smallskip
  \textit{Optimizer Trace (key steps):}
  \begin{verbatim}
"strategy":"ref_or_null" on r (4 rows),
then "unique_subquery" on p and e (1 row each),
filesort for ORDER BY.
  \end{verbatim}

  \item \textbf{Forced Index Only}  
  \begin{verbatim}

FROM Rating      AS r
  FORCE INDEX (uk_rating_once)
JOIN Performance AS p
  FORCE INDEX (PRIMARY)
  ON r.performance_id_FK = p.performance_id
JOIN Event      AS e
  FORCE INDEX (PRIMARY)
  ON p.event_id_FK      = e.event_id
… (ίδιο WHERE/GROUP/ORDER)
  \end{verbatim}
  \textit{EXPLAIN FORMAT=JSON: cost = 3.95, key=r→uk\_rating\_once, p/e→PRIMARY}

  \item \textbf{Forced Nested Loop Join}  
  \begin{verbatim}

FROM Rating      AS r
  FORCE INDEX (uk_rating_once)
JOIN Performance AS p
  FORCE INDEX (PRIMARY)
  /*+ LOOP_JOIN(r p) */
  ON r.performance_id_FK = p.performance_id
JOIN Event      AS e
  FORCE INDEX (PRIMARY)
  ON p.event_id_FK      = e.event_id
… 
  \end{verbatim}
  \textit{EXPLAIN FORMAT=JSON: cost ≃ 3.95, join remains NL}

  \item \textbf{Forced Hash Join}  
  \begin{verbatim}

/*+ HASH_JOIN(r p) */
SELECT …
  (ίδιο FROM/JOIN/WHERE/GROUP/ORDER)
  \end{verbatim}
  \begin{verbatim}
EXPLAIN FORMAT=JSON: cost ≃ 3.95, build_hash_table on p (115 rows), probe_hash on r (4 rows)
  \end{verbatim}

\end{enumerate}

\paragraph{Συμπεράσματα}
\begin{itemize}
  \item Ο optimizer επιλέγει \emph{Nested Loop} με χρήση του \texttt{uk\_rating\_once} index για το πρώτο join (4 rows).  
  \item Forced index ή hints δεν βελτιώνουν το cost, καθώς το σύνολο είναι ήδη μικρό.  
  \item Ο \emph{hash join} δεν κερδίζει λόγω πολύ μικρού πλήθους εγγραφών.  
  \item Η filesort για το \texttt{ORDER BY} παραμένει απαραίτητη.
\end{itemize}

\paragraph{Συμπεράσματα}  
Εδώ το Nested-Loop είναι και πάλι ταχύτερο (~0.12 ms) έναντι των Hash (~0.09 ms) και Merge (~0.15 ms), λόγω του σχετικά μικρού όγκου δεδομένων και του καλού indexing.
